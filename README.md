<!-- ================= SAMPLE ================= -->

<h2>💻 Sample Code (Bubble Sort)</h2>

<pre>
for(int i=0; i&lt;n-1; i++) {
    for(int j=0; j&lt;n-i-1; j++) {
        if(arr[j] &gt; arr[j+1]) {
            int temp = arr[j];
            arr[j] = arr[j+1];
            arr[j+1] = temp;
        }
    }
}
</pre>
